cmake_minimum_required(VERSION 3.24)
include(cmake/AppIdentity.cmake)
project(Mira VERSION ${APP_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ANDROID)
  # Qt WebEngine does not exist on Android; playback there goes through Spotify's own app.
  find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network Concurrent ShaderTools)
else()
  find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network Concurrent ShaderTools WebEngineQuick WebEngineCore WebChannel)
endif()
qt_standard_project_setup(REQUIRES 6.8)
set(QT_QML_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/qml)
add_subdirectory(third_party/m3shapes)
configure_file(src/AppConfig.h.in generated/AppConfig.h @ONLY)
qt_add_executable(mira WIN32 MACOSX_BUNDLE src/main.cpp src/WebPlayback.h src/TokenStore.cpp src/TokenStore.h src/OAuth.cpp src/OAuth.h src/SpotifyApi.cpp src/SpotifyApi.h src/Preferences.cpp src/Preferences.h src/FocusSession.cpp src/FocusSession.h)
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
if(ANDROID)
  target_sources(mira PRIVATE src/WebPlaybackStub.cpp)
else()
  target_sources(mira PRIVATE src/WebPlayback.cpp)
  target_link_libraries(mira PRIVATE Qt6::WebEngineQuick Qt6::WebEngineCore Qt6::WebChannel)
endif()
if(UNIX AND NOT APPLE AND NOT ANDROID)
  # MPRIS media controls for desktop widgets, lock screens and media keys.
  find_package(Qt6 REQUIRED COMPONENTS DBus)
  target_sources(mira PRIVATE src/Mpris.cpp src/Mpris.h)
  target_compile_definitions(mira PRIVATE MIRA_MPRIS)
  target_link_libraries(mira PRIVATE Qt6::DBus)
endif()
qt_add_qml_module(mira URI Mira VERSION 1.0 QML_FILES qml/Main.qml qml/FocusPage.qml qml/MusicPage.qml qml/SettingsPage.qml qml/Surface.qml qml/Theme.qml qml/Icon.qml qml/ActionButton.qml qml/ShapeArt.qml qml/CoverArt.qml qml/PlayerBar.qml qml/WavyProgress.qml qml/LoadingIndicator.qml qml/NowPlaying.qml qml/NavigationBar.qml)
qt_add_resources(mira assets PREFIX / FILES assets/mira.svg assets/mira.png assets/player/index.html assets/player/player.js)
target_include_directories(mira PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(mira PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Concurrent m3shapesplugin)
if(WIN32)
  target_link_libraries(mira PRIVATE Advapi32)
  string(REPLACE "." "," APP_VERSION_COMMAS "${APP_VERSION}")
  configure_file(packaging/windows.rc.in generated/windows.rc @ONLY)
  target_sources(mira PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated/windows.rc)
elseif(APPLE)
  target_link_libraries(mira PRIVATE "-framework Security" "-framework CoreFoundation")
  set_source_files_properties(assets/mira.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  target_sources(mira PRIVATE assets/mira.icns)
endif()
set_target_properties(mira PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "${APP_NAME}" MACOSX_BUNDLE_GUI_IDENTIFIER "${APP_ID}" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_ICON_FILE mira.icns)
if(ANDROID)
  # 0.4.4 -> 404: must increase with every release so Android accepts the update.
  math(EXPR APP_VERSION_CODE "${PROJECT_VERSION_MAJOR} * 10000 + ${PROJECT_VERSION_MINOR} * 100 + ${PROJECT_VERSION_PATCH}")
  set_target_properties(mira PROPERTIES
    QT_ANDROID_PACKAGE_NAME "${APP_ID}"
    QT_ANDROID_VERSION_NAME "${APP_VERSION}"
    QT_ANDROID_VERSION_CODE "${APP_VERSION_CODE}"
    QT_ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android")
  # Qt's HTTPS on Android needs OpenSSL in the APK. KDAB's prebuilt libraries, pinned to one commit.
  include(FetchContent)
  FetchContent_Declare(android_openssl
    DOWNLOAD_EXTRACT_TIMESTAMP true
    URL https://github.com/KDAB/android_openssl/archive/b71f1470962019bd89534a2919f5925f93bc5779.zip)
  FetchContent_MakeAvailable(android_openssl)
  include(${android_openssl_SOURCE_DIR}/android_openssl.cmake)
  add_android_openssl_libraries(mira)
else()
  include(cmake/Packaging.cmake)
endif()
